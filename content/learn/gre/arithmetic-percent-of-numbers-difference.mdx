<Question>
What is $15\%$ of $240$ minus $12\%$ of $150$?
</Question>

<Choices answer="B">
<Choice label="A">16</Choice>
<Choice label="B">18</Choice>
<Choice label="C">20</Choice>
<Choice label="D">22</Choice>
<Choice label="E">24</Choice>
</Choices>
</QuestionPane>

<ExplanationPane>
<Explanation>
<SolutionStep number={1} title="Compute 15% of 240">
$0.15 × 240 = 36$.
</SolutionStep>

<SolutionStep number={2} title="Compute 12% of 150">
$0.12 × 150 = 18$.
</SolutionStep>

<SolutionStep number={3} title="Subtract">
$36 - 18 = 18$, choice **B**.
</SolutionStep>
</Explanation>
